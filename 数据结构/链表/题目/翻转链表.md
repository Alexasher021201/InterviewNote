class Solution {
    public ListNode reverseList(ListNode head) {
        ListNode prev = null;  //用于标记翻转后的链表
        ListNode curr = head;  //用于表示正在处理的链表
        while(curr != null){
            ListNode next = curr.next;  //用于表示剩下的未处理的链表，当前指针对应的链表头会被拿到prev中
            curr.next = prev;  //将prev接到curr的后面
            prev = curr;  //将当前节点添加到prev中
            curr = next;  //将剩下未处理的节点接回到curr，以供下次使用
        }
        return prev;
    }
}
